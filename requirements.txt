pandera>=0.20,<1.0
biopython>=1.83

# The lint gate. CLAUDE.md makes `ruff check` the second of the two
# checks a build has to pass, and README.md tells a reader to install
# from this file, so a gate the documented install cannot run is a gate
# only people with ruff already on their machine have ever run.
ruff>=0.6,<1.0

# tools/verify_references.py only. Title comparison against what Crossref
# returns, so a reference whose title drifted from the record is a score
# rather than an argument. No build imports it and no test needs it.
rapidfuzz>=3.0,<4.0

# tools/generate_qr_codes.py only. qrcode writes the codes and pulls Pillow
# for PNG output. zxing-cpp reads them back.
#
# The decoder is a separate library on purpose, and it is not the same code
# path as the encoder. Decoding with the library that did the encoding would
# check that qrcode agrees with itself, which is not the question: the
# question is whether a scanner that has never heard of qrcode gets the URL
# back. These codes go on paper, and paper is the one medium where a mistake
# cannot be corrected after the fact.
qrcode[pil]>=8.0,<9.0
zxing-cpp>=2.2,<3.0

# Introduced in Chapter 3, after the raw loop
langgraph>=1.0,<2.0
